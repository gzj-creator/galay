function(galay_add_postgres_example target_name source_file)
    add_executable(${target_name} ${source_file})
    target_link_libraries(${target_name} PRIVATE galay::postgres)
    target_include_directories(${target_name} PRIVATE ${CMAKE_BINARY_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR})
    set_target_properties(${target_name} PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
endfunction()

function(galay_add_postgres_import_example target_name source_file)
    if(NOT GALAY_POSTGRES_CPP23_MODULES_EFFECTIVE)
        return()
    endif()
    galay_add_postgres_example(${target_name} ${source_file})
endfunction()

set(GALAY_POSTGRES_EXAMPLE_SCENARIO_e1_query sync_query)
set(GALAY_POSTGRES_EXAMPLE_SCENARIO_e2_async_query async_query)
set(GALAY_POSTGRES_EXAMPLE_SCENARIO_e3_pool pool)
set(GALAY_POSTGRES_EXAMPLE_SCENARIO_e4_prepared prepared_statement)
set(GALAY_POSTGRES_EXAMPLE_SCENARIO_e5_pipeline pipeline)

file(GLOB GALAY_POSTGRES_INCLUDE_EXAMPLES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/*.cc")
foreach(source IN LISTS GALAY_POSTGRES_INCLUDE_EXAMPLES)
    get_filename_component(name "${source}" NAME_WE)
    set(scenario "${name}")
    if(DEFINED GALAY_POSTGRES_EXAMPLE_SCENARIO_${name})
        set(scenario "${GALAY_POSTGRES_EXAMPLE_SCENARIO_${name}}")
    endif()
    galay_add_postgres_example(example_postgres_include_${scenario} ${source})
endforeach()

if(GALAY_POSTGRES_CPP23_MODULES_EFFECTIVE)
    file(GLOB GALAY_POSTGRES_IMPORT_EXAMPLES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/import/*.cc")
    foreach(source IN LISTS GALAY_POSTGRES_IMPORT_EXAMPLES)
        get_filename_component(name "${source}" NAME_WE)
        set(scenario "${name}")
        if(DEFINED GALAY_POSTGRES_EXAMPLE_SCENARIO_${name})
            set(scenario "${GALAY_POSTGRES_EXAMPLE_SCENARIO_${name}}")
        endif()
        galay_add_postgres_import_example(example_postgres_import_${scenario} ${source})
    endforeach()
endif()
