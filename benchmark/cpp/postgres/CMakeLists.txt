function(galay_add_postgres_benchmark target_name source_file)
    add_executable(${target_name} ${source_file})
    target_link_libraries(${target_name} PRIVATE galay::postgres)
    target_include_directories(${target_name} PRIVATE ${CMAKE_BINARY_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR})
    set_target_properties(${target_name} PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
endfunction()

file(GLOB GALAY_POSTGRES_BENCHMARK_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/b*.cc")
foreach(source IN LISTS GALAY_POSTGRES_BENCHMARK_SOURCES)
    get_filename_component(name "${source}" NAME_WE)
    string(REGEX REPLACE "^b[0-9]+_" "" scenario "${name}")
    galay_add_postgres_benchmark(benchmark_postgres_${scenario} ${source})
endforeach()

if(PkgConfig_FOUND)
    pkg_check_modules(GALAY_BENCH_LIBPQ QUIET IMPORTED_TARGET libpq)
endif()
if(TARGET PkgConfig::GALAY_BENCH_LIBPQ)
    find_package(Threads REQUIRED)
    file(GLOB GALAY_LIBPQ_BENCHMARK_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/compare/libpq*.cc")
    foreach(source IN LISTS GALAY_LIBPQ_BENCHMARK_SOURCES)
        get_filename_component(name "${source}" NAME_WE)
        string(REGEX REPLACE "^libpq_" "" scenario "${name}")
        set(target_name "benchmark_postgres_libpq_${scenario}")
        add_executable(${target_name} ${source})
        target_link_libraries(${target_name} PRIVATE PkgConfig::GALAY_BENCH_LIBPQ Threads::Threads)
        target_include_directories(${target_name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/common)
        target_compile_features(${target_name} PRIVATE cxx_std_23)
    endforeach()
else()
    message(STATUS "Skipping benchmark_postgres_libpq_query_pressure: libpq not found")
endif()
