cmake_minimum_required(VERSION 3.28)
project(galay_serde_consumer LANGUAGES CXX)

find_package(galay CONFIG REQUIRED)
if(GALAY_EXPECT_NO_MODULES)
    get_target_property(serde_dependencies galay::serde INTERFACE_LINK_LIBRARIES)
    if(serde_dependencies MATCHES "cpp23-modules" OR TARGET serde::serde-cpp23-modules)
        message(FATAL_ERROR "Header-only installation must not require module providers")
    endif()
endif()
enable_testing()
if(MSVC)
    add_compile_options(/UNDEBUG)
else()
    add_compile_options(-UNDEBUG)
endif()
file(GLOB consumer_sources CONFIGURE_DEPENDS "t*.cc")
list(FILTER consumer_sources EXCLUDE REGEX "/t13_native_modules\\.cc$")
foreach(source IN LISTS consumer_sources)
    get_filename_component(name "${source}" NAME_WE)
    if(name STREQUAL "t10_import_smoke" AND NOT TARGET serde::serde-cpp23-modules)
        continue()
    endif()
    if(name STREQUAL "t12_shared_backend" AND
       (NOT TARGET galay::mcp OR NOT TARGET galay::etcd))
        continue()
    endif()
    add_executable("${name}" "${source}")
    set_target_properties("${name}" PROPERTIES CXX_EXTENSIONS OFF)
    target_link_libraries("${name}" PRIVATE galay::serde)
    if(name STREQUAL "t10_import_smoke")
        set_target_properties("${name}" PROPERTIES CXX_SCAN_FOR_MODULES ON)
    elseif(name STREQUAL "t12_shared_backend")
        target_link_libraries("${name}" PRIVATE galay::mcp galay::etcd)
    endif()
    add_test(NAME "installed.${name}" COMMAND "${name}")
endforeach()

if(TARGET galay::kernel AND TARGET galay::postgres AND TARGET galay::rpc)
    get_target_property(kernel_modules galay::kernel INTERFACE_CXX_MODULE_SETS)
    get_target_property(postgres_modules galay::postgres INTERFACE_CXX_MODULE_SETS)
    get_target_property(rpc_modules galay::rpc INTERFACE_CXX_MODULE_SETS)
    if(kernel_modules AND postgres_modules AND rpc_modules)
        add_executable(t13_native_modules t13_native_modules.cc)
        set_target_properties(t13_native_modules PROPERTIES
            CXX_EXTENSIONS OFF CXX_SCAN_FOR_MODULES ON)
        target_link_libraries(t13_native_modules PRIVATE
            galay::kernel galay::postgres galay::rpc galay::serde)
        add_test(NAME installed.native_modules COMMAND t13_native_modules)
    endif()
endif()
