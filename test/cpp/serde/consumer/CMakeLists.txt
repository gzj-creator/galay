cmake_minimum_required(VERSION 3.28)
project(galay_serde_consumer LANGUAGES CXX)

find_package(galay CONFIG REQUIRED)
enable_testing()
if(MSVC)
    add_compile_options(/UNDEBUG)
else()
    add_compile_options(-UNDEBUG)
endif()
file(GLOB consumer_sources CONFIGURE_DEPENDS "t*.cc")
foreach(source IN LISTS consumer_sources)
    get_filename_component(name "${source}" NAME_WE)
    if(name STREQUAL "t10_import_smoke" AND NOT TARGET galay::serde-modules)
        continue()
    endif()
    if(name STREQUAL "t12_shared_backend" AND
       (NOT TARGET galay::mcp OR NOT TARGET galay::etcd))
        continue()
    endif()
    add_executable("${name}" "${source}")
    set_target_properties("${name}" PROPERTIES CXX_EXTENSIONS OFF)
    target_link_libraries("${name}" PRIVATE galay::serde)
    if(name STREQUAL "t10_import_smoke")
        target_link_libraries("${name}" PRIVATE galay::serde-modules)
        set_target_properties("${name}" PROPERTIES CXX_SCAN_FOR_MODULES ON)
    elseif(name STREQUAL "t12_shared_backend")
        target_link_libraries("${name}" PRIVATE galay::mcp galay::etcd)
    endif()
    add_test(NAME "installed.${name}" COMMAND "${name}")
endforeach()
